3.2 Dispatch Types

type EngineName = 'memory_core' | 'learning_engine' | 'imagination_engine' 
                | 'reasoning_engine' | 'governance_guard';

interface DispatchOptions {
  timeout?: number;        // ms (default: 30000, long ops: 60000)
  retries?: number;        // default: 0
  retryDelay?: number;     // ms (default: 1000)
  deterministicMode?: boolean;
  priority?: number;
  chainContext?: ChainContext;
}

interface DispatchResult<T> {
  success: boolean;
  engine: EngineName;
  command: string;
  data?: T;
  error?: string;
  errorCode?: DispatchErrorCode;
  startTime: string;
  endTime: string;
  durationMs: number;
  retryCount: number;
  stage: DispatchStage;
}

type DispatchStage = 'pending' | 'routing' | 'executing' | 'completed' | 'failed';
type DispatchErrorCode = 'TIMEOUT' | 'ENGINE_NOT_FOUND' | 'COMMAND_NOT_FOUND' 
                       | 'EXECUTION_FAILED' | 'RETRY_EXHAUSTED' | 'CHAIN_BROKEN' 
                       | 'GOVERNANCE_BLOCKED';

4.2 Hook Usage Pattern

import { useCore, useBrain, useMemoryModule } from '@/hooks/substrate';

function MyComponent() {
  const core = useCore();
  const brain = useBrain();
  const memory = useMemoryModule();

  // Check system health
  if (core.isBooted && brain.health > 70) {
    // System is operational
  }

  // Store a memory
  const handleSave = async () => {
    await memory.ingest('Important insight', { tags: ['discovery'] });
  };

  // Recall memories
  const handleRecall = async () => {
    const results = await memory.retrieve({ query: 'insight', limit: 5 });
  };
}

6. Error Handling Strategy

Graceful Degradation

  • Every module boot is wrapped in try/catch — failures are logged but don't prevent other modules from booting
  • Engine Bus load-sheds at capacity with clear error messages
  • Persistence failures trigger retry with exponential backoff — never crash
  • Circuit breakers automatically isolate failing modules

Error Codes

CodeMeaningRecovery
TIMEOUTOperation exceeded timeoutRetry with longer timeout
ENGINE_NOT_FOUNDTarget engine doesn't existCheck engine name
COMMAND_NOT_FOUNDCommand not in routing mapCheck command name
EXECUTION_FAILEDHandler threw an errorCheck error message
